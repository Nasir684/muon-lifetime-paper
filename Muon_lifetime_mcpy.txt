# -*- coding: utf-8 -*- # muon_lifetime_mc.py # Monte-Carlo validation pipeline for the positive-muon lifetime measurement. # Implements the statistical protocol of Khan (2026), Sec. 4: # - truncated-exponential + flat-background generative model # - unbinned extended maximum-likelihood fit (tau, eps_b) # - profile-likelihood interval (Wilks, dlnL = 0.5 for 68%) # - Fano statistic F = mean(z^2) with sqrt(2/J) uncertainty # - parametric-bootstrap (Lilliefors-corrected) KS test # - N_exp pseudo-experiment bias and coverage validation import numpy as np from scipy.optimize import minimize TAU_PDG = 2.1969811 # us, PDG 2022 T_MIN = 0.5 # us, example analysis window T_MAX = 15.0 # us EPS_B = 0.02 # flat accidental fraction, example def znorm(tau, t_min, t_max): return np.exp(-t_min/tau) - np.exp(-t_max/tau) def pdf(t, tau, eps_b, t_min, t_max): Z = tau * znorm(tau, t_min, t_max) W = t_max - t_min return (1.0 - eps_b) * np.exp(-t/tau) / Z + eps_b / W def cdf(t, tau, eps_b, t_min, t_max): Zn = znorm(tau, t_min, t_max) W = t_max - t_min return ((1.0 - eps_b) * (np.exp(-t_min/tau) - np.exp(-t/tau)) / Zn + eps_b * (t - t_min) / W) def _negloglik(params, t, t_min, t_max): ln_tau, x = params tau = np.exp(ln_tau) eps_b = 1.0 / (1.0 + np.exp(-x)) return -np.sum(np.log(pdf(t, tau, eps_b, t_min, t_max))) def fit_spectrum(t, t_min=T_MIN, t_max=T_MAX, tau0=2.2, eps0=0.02): x0 = np.array([np.log(tau0), np.log(eps0/(1.0 - eps0))]) res = minimize(_negloglik, x0, args=(t, t_min, t_max), method="BFGS") ln_tau, x = res.x tau_hat = np.exp(ln_tau) eps_hat = 1.0 / (1.0 + np.exp(-x)) try: sigma_tau = tau_hat * np.sqrt(res.hess_inv[0, 0]) except Exception: sigma_tau = np.nan return tau_hat, eps_hat, sigma_tau, res.fun, res def profile_interval(t, tau_hat, eps_hat, negll_best, t_min=T_MIN, t_max=T_MAX, half_width=0.02, n_grid=161, dlnl=0.5): grid = np.linspace(tau_hat*(1.0 - half_width), tau_hat*(1.0 + half_width), n_grid) x0 = np.array([np.log(eps_hat/(1.0 - eps_hat))]) dll = np.empty(n_grid) for i, tv in enumerate(grid): r = minimize(lambda x: _negloglik([np.log(tv), x[0]], t, t_min, t_max), x0, method="Nelder-Mead", options={"xatol": 1e-8, "fatol": 1e-8, "maxiter": 200}) dll[i] = r.fun - negll_best lo = _crossing(grid, dll, dlnl, "lo") hi = _crossing(grid, dll, dlnl, "hi") return lo, hi, grid, dll def _crossing(grid, dll, level, side): i0 = int(np.argmin(dll)) if side == "lo": idx = np.where((grid < grid[i0]) & (dll >= level))[0] if len(idx) == 0: return np.nan i, j = idx[-1], idx[-1] + 1 else: idx = np.where((grid > grid[i0]) & (dll >= level))[0] if len(idx) == 0: return np.nan i, j = idx[0], idx[0] - 1 g1, g2 = grid[i], grid[j] d1, d2 = dll[i], dll[j] return g1 + (level - d1) * (g2 - g1) / (d2 - d1) def sample_times(n, tau, eps_b, t_min, t_max, rng): n_b = rng.binomial(n, eps_b) n_s = n - n_b u = rng.random(n_s) Zn = znorm(tau, t_min, t_max) ts = -tau * np.log(np.exp(-t_min/tau) - u * Zn) tb = rng.uniform(t_min, t_max, n_b) return np.concatenate([ts, tb]) def fano_statistic(t, tau, eps, t_min=T_MIN, t_max=T_MAX, J=40): edges = np.linspace(t_min, t_max, J + 1) n, _ = np.histogram(t, bins=edges) mu = len(t) * (cdf(edges[1:], tau, eps, t_min, t_max) - cdf(edges[:-1], tau, eps, t_min, t_max)) z = (n - mu) / np.sqrt(mu) return np.mean(z**2), np.sqrt(2.0/J), n, mu def ks_D(t, tau, eps, t_min=T_MIN, t_max=T_MAX): xs = cdf(np.sort(t), tau, eps, t_min, t_max) n = len(xs) return max(np.max(np.abs(np.arange(1, n+1)/n - xs)), np.max(np.abs(xs - np.arange(0, n)/n))) def bootstrap_ks(t, tau_hat, eps_hat, t_min=T_MIN, t_max=T_MAX, M=300, seed=None): rng = np.random.default_rng(seed) D_obs = ks_D(t, tau_hat, eps_hat, t_min, t_max) n = len(t) exceed = 0 for _ in range(M): ts = sample_times(rng.poisson(n), tau_hat, eps_hat, t_min, t_max, rng) tb, eb, _, _, _ = fit_spectrum(ts, t_min, t_max) if ks_D(ts, tb, eb, t_min, t_max) >= D_obs: exceed += 1 return (exceed + 1)/(M + 1), D_obs, exceed def mc_validate(n_exp=200, n_events=15000, tau_true=TAU_PDG, eps_b=EPS_B, t_min=T_MIN, t_max=T_MAX, seed=123): rng = np.random.default_rng(seed) tau_hats, sigmas, cover = [], [], 0 for _ in range(n_exp): t = sample_times(rng.poisson(n_events), tau_true, eps_b, t_min, t_max, rng) th, _, sg, _, _ = fit_spectrum(t, t_min, t_max) tau_hats.append(th) sigmas.append(sg) if abs(th - tau_true) <= sg: cover += 1 tau_hats = np.array(tau_hats) sigmas = np.array(sigmas) return dict(n_exp=n_exp, mean_tau=np.mean(tau_hats), bias=np.mean(tau_hats) - tau_true, bias_in_sigma=(np.mean(tau_hats) - tau_true)/np.mean(sigmas), mean_sigma=np.mean(sigmas), coverage=cover/n_exp, sigma_mc=np.std(tau_hats)) def demo(): rng = np.random.default_rng(2026) print("=== single-spectrum demo ===") t = sample_times(rng.poisson(40000), TAU_PDG, EPS_B, T_MIN, T_MAX, rng) th, eh, sg, negll, _ = fit_spectrum(t) print("fitted tau = %.6f us (true %.6f)" % (th, TAU_PDG)) print("eps_b = %.4f" % eh) print("sigma_tau (Fisher) = %.6f us (rel %.4f%%)" % (sg, 100*sg/th)) lo, hi, grid, dll = profile_interval(t, th, eh, negll) print("profile 68%% CI = [%.6f, %.6f] us" % (lo, hi)) F, sF, n, mu = fano_statistic(t, th, eh) print("Fano F = %.3f +- %.3f (Poisson: F=1)" % (F, sF)) p, D, ex = bootstrap_ks(t, th, eh, M=300, seed=7) print("bootstrap-KS: D = %.4f, p = %.3f (M=300 demo)" % (D, p)) print("=== MC validation (reduced settings) ===") for k, v in mc_validate(n_exp=200, n_events=15000).items(): print(" %s = %s" % (k, v)) if __name__ == "__main__": demo()
